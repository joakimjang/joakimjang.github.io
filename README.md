document.addEventListener('DOMContentLoaded', () => {
  const menuButton = document.querySelector('.menu-btn');
  const nav = document.querySelector('.main-nav');

  if (menuButton && nav) {
    menuButton.addEventListener('click', () => {
      nav.classList.toggle('mobile-open');
    });
  }

  const form = document.getElementById('inquiry-form');
  const summary = document.getElementById('form-summary');

  if (form && summary) {
    form.addEventListener('submit', (event) => {
      event.preventDefault();

      const formData = new FormData(form);
      const businessDirection = formData.get('businessDirection') || '선택하지 않음';
      const partnerType = formData.get('partnerType') || '미입력';
      const industry = formData.get('industry') || '미입력';
      const market = formData.get('market') || '미입력';
      const services = formData.getAll('service');
      const requirements = formData.get('requirements') || '미입력';
      const message = formData.get('message') || '미입력';

      const selectedServices = services.length ? services.join(', ') : '선택 없음';

      summary.innerHTML = [
        '문의 내용이 정리되었습니다.',
        '',
        `사업 방향: ${businessDirection}`,
        `찾는 파트너 유형: ${partnerType}`,
        `제품·기술·산업: ${industry}`,
        `목표 국가·시장: ${market}`,
        `필요한 서비스: ${selectedServices}`,
        `요구사항: ${requirements}`,
        `문의 내용: ${message}`
      ].join('\n');
    });
  }
});
